📖 ¿Qué es la POO?

La Programación Orientada a Objetos organiza el código en "objetos" que representan cosas del mundo real.

🔹 Tu Primera Clase

// Definición de la clase
class Persona
{
    // Atributos (campos)
    public string nombre;
    public int edad;
    
    // Método
    public void Presentarse()
    {
        Console.WriteLine($"Hola, soy {nombre} y tengo {edad} años.");
    }
}

// Crear y usar objetos
Persona persona1 = new Persona();
persona1.nombre = "Ana";
persona1.edad = 25;
persona1.Presentarse();  // Hola, soy Ana y tengo 25 años.

Persona persona2 = new Persona();
persona2.nombre = "Carlos";
persona2.edad = 30;
persona2.Presentarse();

🔹 Constructores

class Producto
{
    public string nombre;
    public double precio;
    public int stock;
    
    // Constructor por defecto
    public Producto()
    {
        nombre = "Sin nombre";
        precio = 0;
        stock = 0;
    }
    
    // Constructor con parámetros
    public Producto(string nombre, double precio, int stock)
    {
        this.nombre = nombre;  // 'this' = objeto actual
        this.precio = precio;
        this.stock = stock;
    }
}

// Uso
Producto p1 = new Producto();  // Usa constructor por defecto
Producto p2 = new Producto("Laptop", 15000, 5);  // Con parámetros

🔹 Encapsulamiento y Propiedades

class CuentaBancaria
{
    // Campo privado
    private decimal saldo;
    
    // Propiedad de solo lectura
    public decimal Saldo
    {
        get { return saldo; }
    }
    
    // Propiedad con validación
    public string Titular { get; set; }
    
    public CuentaBancaria(string titular)
    {
        Titular = titular;
        saldo = 0;
    }
    
    public bool Depositar(decimal cantidad)
    {
        if (cantidad <= 0) return false;
        saldo += cantidad;
        return true;
    }
    
    public bool Retirar(decimal cantidad)
    {
        if (cantidad <= 0 || cantidad > saldo) return false;
        saldo -= cantidad;
        return true;
    }
}

// Uso
CuentaBancaria cuenta = new CuentaBancaria("Juan");
cuenta.Depositar(1000);
cuenta.Retirar(200);
Console.WriteLine($"Saldo: ${cuenta.Saldo}");  // 800
// cuenta.saldo = 1000000;  // ERROR: saldo es privado

🔹 Herencia

// Clase base
class Animal
{
    public string Nombre { get; set; }
    
    public virtual void HacerSonido()
    {
        Console.WriteLine("El animal hace un sonido");
    }
}

// Clase derivada
class Perro : Animal
{
    public string Raza { get; set; }
    
    public override void HacerSonido()
    {
        Console.WriteLine($"{Nombre} dice: ¡Guau guau!");
    }
    
    public void BuscarPelota()
    {
        Console.WriteLine($"{Nombre} busca la pelota");
    }
}

class Gato : Animal
{
    public override void HacerSonido()
    {
        Console.WriteLine($"{Nombre} dice: ¡Miau!");
    }
}

// Uso
Perro firulais = new Perro { Nombre = "Firulais", Raza = "Labrador" };
Gato michi = new Gato { Nombre = "Michi" };

firulais.HacerSonido();  // Firulais dice: ¡Guau guau!
michi.HacerSonido();     // Michi dice: ¡Miau!

// Polimorfismo
Animal[] animales = { firulais, michi };
foreach (Animal a in animales)
{
    a.HacerSonido();  // Cada uno hace su propio sonido
}

🔹 Miembros Estáticos

class Calculadora
{
    // Métodos estáticos - se llaman sin crear objeto
    public static int Sumar(int a, int b) => a + b;
    public static int Restar(int a, int b) => a - b;
}

// Uso directo sin instanciar
int suma = Calculadora.Sumar(5, 3);  // 8

class Contador
{
    // Campo estático compartido por todas las instancias
    private static int total = 0;
    
    public int Id { get; }
    
    public Contador()
    {
        total++;
        Id = total;
    }
    
    public static int ObtenerTotal() => total;
}

Contador c1 = new Contador();  // Id = 1
Contador c2 = new Contador();  // Id = 2
Console.WriteLine(Contador.ObtenerTotal());  // 2
