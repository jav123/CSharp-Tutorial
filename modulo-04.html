📖 ¿Qué es una Función?

Una función (o método) es un bloque de código reutilizable que realiza una tarea específica.

Beneficios de usar funciones:

BeneficioDescripción
ReutilizaciónEscribes una vez, usas muchas veces
OrganizaciónDivide problemas grandes en partes pequeñas
MantenimientoCambias en un solo lugar
LegibilidadCódigo más fácil de entender

🔹 Función Básica (void)

// Función que no retorna valor
static void Saludar()
{
    Console.WriteLine("¡Hola! Bienvenido al programa.");
}

// Llamar a la función
static void Main()
{
    Saludar();  // Imprime: ¡Hola! Bienvenido al programa.
    Saludar();  // Puedes llamarla múltiples veces
}

🔹 Funciones con Parámetros

// Un parámetro
static void SaludarPersona(string nombre)
{
    Console.WriteLine($"¡Hola, {nombre}!");
}

// Múltiples parámetros
static void MostrarInfo(string nombre, int edad, string ciudad)
{
    Console.WriteLine($"Nombre: {nombre}");
    Console.WriteLine($"Edad: {edad} años");
    Console.WriteLine($"Ciudad: {ciudad}");
}

// Uso
SaludarPersona("Ana");
SaludarPersona("Carlos");
MostrarInfo("María", 25, "México");

🔹 Funciones con Valor de Retorno

// Función que retorna un entero
static int Sumar(int a, int b)
{
    return a + b;
}

// Función que retorna un string
static string CrearSaludo(string nombre)
{
    return $"¡Bienvenido, {nombre}!";
}

// Función que retorna un booleano
static bool EsPar(int numero)
{
    return numero % 2 == 0;
}

// Uso
int resultado = Sumar(5, 3);  // resultado = 8
Console.WriteLine(CrearSaludo("Ana"));
Console.WriteLine($"¿8 es par? {EsPar(8)}");  // true

🔹 Calculadora con Funciones

class Calculadora
{
    static void Main()
    {
        double a = 10, b = 3;
        
        Console.WriteLine($"Suma: {Sumar(a, b)}");
        Console.WriteLine($"Resta: {Restar(a, b)}");
        Console.WriteLine($"Multiplicación: {Multiplicar(a, b)}");
        Console.WriteLine($"División: {Dividir(a, b)}");
    }
    
    static double Sumar(double a, double b) => a + b;
    static double Restar(double a, double b) => a - b;
    static double Multiplicar(double a, double b) => a * b;
    static double Dividir(double a, double b) => b != 0 ? a / b : 0;
}

🔹 Parámetros Opcionales

static void ImprimirMensaje(string mensaje, int veces = 1, string prefijo = "-")
{
    for (int i = 0; i < veces; i++)
    {
        Console.WriteLine($"{prefijo} {mensaje}");
    }
}

// Diferentes formas de llamar
ImprimirMensaje("Hola");              // Usa valores por defecto
ImprimirMensaje("Hola", 3);           // Repite 3 veces
ImprimirMensaje("Hola", 2, "*");     // Personaliza todo
ImprimirMensaje("Hola", prefijo: ">"); // Parámetro nombrado

🔹 Parámetros ref y out

// ref: modifica la variable original
static void Duplicar(ref int numero)
{
    numero *= 2;
}

int valor = 5;
Duplicar(ref valor);
Console.WriteLine(valor);  // 10

// out: retorna múltiples valores
static void Dividir(int a, int b, out int cociente, out int resto)
{
    cociente = a / b;
    resto = a % b;
}

Dividir(17, 5, out int c, out int r);
Console.WriteLine($"17 ÷ 5 = {c} resto {r}");  // 3 resto 2

🔹 Sobrecarga de Métodos

// Mismo nombre, diferentes parámetros
static int Sumar(int a, int b) => a + b;
static int Sumar(int a, int b, int c) => a + b + c;
static double Sumar(double a, double b) => a + b;

// C# elige automáticamente la versión correcta
Console.WriteLine(Sumar(5, 3));         // 8 (int, int)
Console.WriteLine(Sumar(5, 3, 2));      // 10 (int, int, int)
Console.WriteLine(Sumar(5.5, 3.3));    // 8.8 (double, double)

🔹 Funciones Recursivas

// Factorial: 5! = 5 × 4 × 3 × 2 × 1 = 120
static long Factorial(int n)
{
    if (n <= 1)
        return 1;  // Caso base
    
    return n * Factorial(n - 1);  // Caso recursivo
}

// Fibonacci: 1, 1, 2, 3, 5, 8, 13...
static int Fibonacci(int n)
{
    if (n <= 2)
        return 1;
    
    return Fibonacci(n - 1) + Fibonacci(n - 2);
}

Console.WriteLine($"5! = {Factorial(5)}");  // 120
Console.WriteLine($"Fib(10) = {Fibonacci(10)}");  // 55
