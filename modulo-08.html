🟢 Ejercicio 1: Calculadora Completa

using System;

class Calculadora
{
    static void Main()
    {
        bool continuar = true;
        
        while (continuar)
        {
            Console.WriteLine("╔═══════════════════════════════╗");
            Console.WriteLine("║     CALCULADORA               ║");
            Console.WriteLine("╠═══════════════════════════════╣");
            Console.WriteLine("║  1. Suma                      ║");
            Console.WriteLine("║  2. Resta                     ║");
            Console.WriteLine("║  3. Multiplicación            ║");
            Console.WriteLine("║  4. División                  ║");
            Console.WriteLine("║  0. Salir                     ║");
            Console.WriteLine("╚═══════════════════════════════╝");
            
            Console.Write("Opción: ");
            string opcion = Console.ReadLine();
            
            if (opcion == "0") { continuar = false; continue; }
            
            Console.Write("Primer número: ");
            double a = double.Parse(Console.ReadLine());
            
            Console.Write("Segundo número: ");
            double b = double.Parse(Console.ReadLine());
            
            double resultado = opcion switch
            {
                "1" => a + b,
                "2" => a - b,
                "3" => a * b,
                "4" => b != 0 ? a / b : 0,
                _ => 0
            };
            
            Console.WriteLine($"\nResultado: {resultado}\n");
        }
    }
}

🟡 Ejercicio 2: Juego de Adivinanza

using System;

class JuegoAdivinanza
{
    static void Main()
    {
        Random random = new Random();
        int numeroSecreto = random.Next(1, 101);
        int intentos = 0;
        int maxIntentos = 7;
        bool adivinado = false;
        
        Console.WriteLine("🎮 Adivina el número (1-100)");
        Console.WriteLine($"Tienes {maxIntentos} intentos.\n");
        
        while (intentos < maxIntentos && !adivinado)
        {
            Console.Write($"Intento {intentos + 1}: ");
            
            if (!int.TryParse(Console.ReadLine(), out int intento))
            {
                Console.WriteLine("Ingresa un número válido.\n");
                continue;
            }
            
            intentos++;
            
            if (intento == numeroSecreto)
            {
                adivinado = true;
                Console.WriteLine($"\n🎉 ¡CORRECTO! Lo lograste en {intentos} intentos.");
            }
            else if (intento < numeroSecreto)
            {
                Console.WriteLine("📈 El número es MAYOR.\n");
            }
            else
            {
                Console.WriteLine("📉 El número es MENOR.\n");
            }
        }
        
        if (!adivinado)
        {
            Console.WriteLine($"\n😢 Se acabaron los intentos. Era: {numeroSecreto}");
        }
    }
}

🟡 Ejercicio 3: Gestor de Tareas

using System;
using System.Collections.Generic;

class Tarea
{
    public int Id { get; set; }
    public string Titulo { get; set; }
    public bool Completada { get; set; }
}

class GestorTareas
{
    static List<Tarea> tareas = new List<Tarea>();
    static int contadorId = 0;
    
    static void Main()
    {
        bool continuar = true;
        
        while (continuar)
        {
            Console.WriteLine("\n═══ GESTOR DE TAREAS ═══");
            Console.WriteLine("1. Ver tareas");
            Console.WriteLine("2. Agregar tarea");
            Console.WriteLine("3. Completar tarea");
            Console.WriteLine("4. Eliminar tarea");
            Console.WriteLine("0. Salir");
            Console.Write("Opción: ");
            
            switch (Console.ReadLine())
            {
                case "1": VerTareas(); break;
                case "2": AgregarTarea(); break;
                case "3": CompletarTarea(); break;
                case "4": EliminarTarea(); break;
                case "0": continuar = false; break;
            }
        }
    }
    
    static void VerTareas()
    {
        if (tareas.Count == 0)
        {
            Console.WriteLine("No hay tareas.");
            return;
        }
        
        foreach (var t in tareas)
        {
            string estado = t.Completada ? "✅" : "⏳";
            Console.WriteLine($"{estado} [{t.Id}] {t.Titulo}");
        }
    }
    
    static void AgregarTarea()
    {
        Console.Write("Título: ");
        string titulo = Console.ReadLine();
        
        tareas.Add(new Tarea 
        { 
            Id = ++contadorId, 
            Titulo = titulo, 
            Completada = false 
        });
        
        Console.WriteLine("✓ Tarea agregada.");
    }
    
    static void CompletarTarea()
    {
        Console.Write("ID de la tarea: ");
        if (int.TryParse(Console.ReadLine(), out int id))
        {
            var tarea = tareas.Find(t => t.Id == id);
            if (tarea != null)
            {
                tarea.Completada = true;
                Console.WriteLine("✓ Tarea completada.");
            }
        }
    }
    
    static void EliminarTarea()
    {
        Console.Write("ID de la tarea: ");
        if (int.TryParse(Console.ReadLine(), out int id))
        {
            tareas.RemoveAll(t => t.Id == id);
            Console.WriteLine("✓ Tarea eliminada.");
        }
    }
}
