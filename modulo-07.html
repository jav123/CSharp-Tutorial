📖 ¿Qué es una Excepción?

Una excepción es un error que ocurre durante la ejecución. Sin manejo adecuado, el programa se detiene.

Excepciones comunes:

ExcepciónCausa
DivideByZeroExceptionDividir entre cero
FormatExceptionConvertir texto inválido
IndexOutOfRangeExceptionÍndice fuera de rango
NullReferenceExceptionAcceder a objeto null
FileNotFoundExceptionArchivo no existe

🔹 Try-Catch Básico

try
{
    Console.Write("Ingresa un número: ");
    int num1 = int.Parse(Console.ReadLine());
    
    Console.Write("Ingresa otro número: ");
    int num2 = int.Parse(Console.ReadLine());
    
    int resultado = num1 / num2;
    Console.WriteLine($"Resultado: {resultado}");
}
catch (Exception ex)
{
    Console.WriteLine($"Error: {ex.Message}");
}

Console.WriteLine("El programa continúa...");

🔹 Múltiples Catch

try
{
    Console.Write("Número: ");
    int numero = int.Parse(Console.ReadLine());
    
    Console.Write("Divisor: ");
    int divisor = int.Parse(Console.ReadLine());
    
    int resultado = numero / divisor;
    
    int[] arr = { 1, 2, 3 };
    Console.Write("Índice (0-2): ");
    int idx = int.Parse(Console.ReadLine());
    Console.WriteLine(arr[idx]);
}
catch (FormatException)
{
    Console.WriteLine("Error: Ingresa un número válido.");
}
catch (DivideByZeroException)
{
    Console.WriteLine("Error: No puedes dividir entre cero.");
}
catch (IndexOutOfRangeException)
{
    Console.WriteLine("Error: Índice fuera de rango.");
}
catch (Exception ex)
{
    Console.WriteLine($"Error inesperado: {ex.Message}");
}

🔹 Finally

El bloque finally siempre se ejecuta, haya o no excepción.

try
{
    Console.WriteLine("Dentro del try");
    int x = 10 / 0;  // Causa excepción
}
catch (Exception)
{
    Console.WriteLine("Dentro del catch");
}
finally
{
    Console.WriteLine("Dentro del finally (siempre se ejecuta)");
}

// Útil para liberar recursos
StreamReader archivo = null;
try
{
    archivo = new StreamReader("datos.txt");
    string contenido = archivo.ReadToEnd();
}
catch (FileNotFoundException)
{
    Console.WriteLine("Archivo no encontrado.");
}
finally
{
    archivo?.Close();  // Siempre cerrar el archivo
}

🔹 Lanzar Excepciones (throw)

static void ValidarEdad(int edad)
{
    if (edad < 0)
    {
        throw new ArgumentException("La edad no puede ser negativa.");
    }
    
    if (edad < 18)
    {
        throw new ArgumentException("Debes ser mayor de 18 años.");
    }
    
    Console.WriteLine("Edad válida.");
}

// Uso
try
{
    ValidarEdad(15);
}
catch (ArgumentException ex)
{
    Console.WriteLine($"Error de validación: {ex.Message}");
}

🔹 Excepciones Personalizadas

// Definir excepción personalizada
class SaldoInsuficienteException : Exception
{
    public decimal SaldoActual { get; }
    public decimal MontoSolicitado { get; }
    
    public SaldoInsuficienteException(decimal saldo, decimal monto)
        : base($"Saldo insuficiente. Tienes ${saldo}, necesitas ${monto}")
    {
        SaldoActual = saldo;
        MontoSolicitado = monto;
    }
}

// Usar la excepción
class CuentaBancaria
{
    public decimal Saldo { get; private set; } = 1000;
    
    public void Retirar(decimal monto)
    {
        if (monto > Saldo)
        {
            throw new SaldoInsuficienteException(Saldo, monto);
        }
        Saldo -= monto;
    }
}

// Manejo
try
{
    var cuenta = new CuentaBancaria();
    cuenta.Retirar(5000);
}
catch (SaldoInsuficienteException ex)
{
    Console.WriteLine(ex.Message);
    Console.WriteLine($"Te faltan: ${ex.MontoSolicitado - ex.SaldoActual}");
}

🔹 TryParse - Evitar Excepciones

// En lugar de int.Parse que puede fallar...
int numero;
bool exito = int.TryParse("123", out numero);
// exito = true, numero = 123

exito = int.TryParse("abc", out numero);
// exito = false, numero = 0

// Uso práctico
Console.Write("Ingresa tu edad: ");
if (int.TryParse(Console.ReadLine(), out int edad))
{
    Console.WriteLine($"Tienes {edad} años.");
}
else
{
    Console.WriteLine("Edad inválida.");
}
